function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // Sheet 1: Candidates var candidateSheet = ss.getSheetByName("Sheet1") || ss.getSheets()[0]; var candidateData = getSheetDataAsObjects(candidateSheet); // Sheet 2: Tasks var taskSheet = ss.getSheetByName("Tasks"); if (!taskSheet) { taskSheet = ss.insertSheet("Tasks"); taskSheet.appendRow(["id", "title", "heading", "date", "completed"]); } var taskData = getSheetDataAsObjects(taskSheet); // Sheet 3: Positions var positionSheet = ss.getSheetByName("Positions"); if (!positionSheet) { positionSheet = ss.insertSheet("Positions"); positionSheet.appendRow(["id", "title", "exp", "maxSalary", "department", "businessUnit", "reportingTo", "createdDate", "status"]); } var positionData = getSheetDataAsObjects(positionSheet); var responseObj = { candidates: candidateData, tasks: taskData, positions: positionData }; return ContentService.createTextOutput(JSON.stringify(responseObj)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); if (data.type === 'position') { var positionSheet = ss.getSheetByName("Positions"); if (!positionSheet) { positionSheet = ss.insertSheet("Positions"); positionSheet.appendRow(["id", "title", "exp", "maxSalary", "department", "businessUnit", "reportingTo", "createdDate", "status"]); } upsertPosition(positionSheet, data); } else if (data.type === 'task' || data.action === 'save_task' || data.id) { var taskSheet = ss.getSheetByName("Tasks"); if (!taskSheet) { taskSheet = ss.insertSheet("Tasks"); taskSheet.appendRow(["id", "title", "heading", "date", "completed"]); } if (data.action === 'delete_task') { deleteRowById(taskSheet, data.id); } else { upsertTask(taskSheet, data); } } else { var candidateSheet = ss.getSheetByName("Sheet1") || ss.getSheets()[0]; updateCandidateStatus(candidateSheet, data); } return ContentService.createTextOutput(JSON.stringify({status: "success"})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function getSheetDataAsObjects(sheet) { var rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; var headers = rows[0]; var data = []; for (var i = 1; i < rows.length; i++) { var obj = {}; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = rows[i][j]; } data.push(obj); } return data; } function upsertPosition(sheet, data) { var rows = sheet.getDataRange().getValues(); if (rows.length === 0) { sheet.appendRow(["id", "title", "exp", "maxSalary", "department", "businessUnit", "reportingTo", "createdDate", "status"]); rows = sheet.getDataRange().getValues(); } var headers = rows[0]; var idCol = headers.indexOf("id"); if (idCol === -1) idCol = 0; for (var i = 1; i < rows.length; i++) { if (rows[i][idCol].toString() === data.id.toString()) { sheet.getRange(i + 1, 1, 1, 9).setValues([[ data.id, data.title, data.exp, data.maxSalary, data.department, data.businessUnit, data.reportingTo, data.createdDate, data.status ]]); return; } } sheet.appendRow([data.id, data.title, data.exp, data.maxSalary, data.department, data.businessUnit, data.reportingTo, data.createdDate, data.status]); } function upsertTask(sheet, data) { var rows = sheet.getDataRange().getValues(); if (rows.length === 0) { sheet.appendRow(["id", "title", "heading", "date", "completed"]); rows = sheet.getDataRange().getValues(); } var headers = rows[0]; var idCol = headers.indexOf("id"); if (idCol === -1) idCol = 0; for (var i = 1; i < rows.length; i++) { if (rows[i][idCol].toString() === data.id.toString()) { sheet.getRange(i + 1, 1, 1, 5).setValues([[ data.id, data.title, data.heading, data.date, data.completed === true || data.completed === 'TRUE' ? 'TRUE' : 'FALSE' ]]); return; } } sheet.appendRow([data.id, data.title, data.heading, data.date, data.completed === true || data.completed === 'TRUE' ? 'TRUE' : 'FALSE']); } function deleteRowById(sheet, id) { var rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return; var headers = rows[0]; var idCol = headers.indexOf("id"); if (idCol === -1) idCol = 0; for (var i = 1; i < rows.length; i++) { if (rows[i][idCol].toString() === id.toString()) { sheet.deleteRow(i + 1); return; } } } function updateCandidateStatus(sheet, data) { var rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return; var headers = rows[0]; var nameCol = headers.indexOf("Candidate Name"); if (nameCol === -1) nameCol = headers.indexOf("NAME"); if (nameCol === -1) nameCol = 0; var statusCol = headers.indexOf("Status"); for (var i = 1; i < rows.length; i++) { if (rows[i][nameCol].toString().toLowerCase() === data.candidateName.toLowerCase()) { if (statusCol !== -1) { sheet.getRange(i + 1, statusCol + 1).setValue(data.status); } break; } } }
